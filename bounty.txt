-- EXECUTE | PC + MOBILE

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local LP = Players.LocalPlayer
getgenv().AutoFollow = true

-- ================= VAR =================
local Char, HRP, Hum
local BV, BG
local CurrentTarget
local LastDrop = {}

-- ================= CONFIG =================
local CFG = {
    FastSpeed = 400,
    MidSpeed  = 400,
    SlowSpeed = 400,

    Near = 25,
    MidDist = 160,

    HeadHeight = 8,
    DropCooldown = 5,

    SquareStepTime = 0.5, -- thời gian giữ mỗi cạnh vuông
}

-- ================= SETUP CHARACTER =================
local function SetupChar(c)
    Char = c
    HRP = c:WaitForChild("HumanoidRootPart")
    Hum = c:WaitForChild("Humanoid")

    if BV then BV:Destroy() end
    if BG then BG:Destroy() end

    BV = Instance.new("BodyVelocity")
    BV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    BV.Velocity = Vector3.zero
    BV.Parent = HRP

    BG = Instance.new("BodyGyro")
    BG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    BG.P = 2e4
    BG.CFrame = HRP.CFrame
    BG.Parent = HRP
end

if LP.Character then SetupChar(LP.Character) end
LP.CharacterAdded:Connect(SetupChar)

-- ================= TARGET CHECK =================
local function InvalidTarget(plr)
    if not plr or plr == LP then return true end
    if not plr.Character then return true end
    if plr.Character:FindFirstChild("ForceField") then return true end

    local h = plr.Character:FindFirstChildOfClass("Humanoid")
    if not h or h.Health <= 0 then return true end

    if LastDrop[plr] and tick() - LastDrop[plr] < CFG.DropCooldown then
        return true
    end
    return false
end

-- ================= FIND TARGET =================
local function GetTarget()
    if not HRP then return nil end
    local best, dist = nil, math.huge

    for _, p in pairs(Players:GetPlayers()) do
        if not InvalidTarget(p) and p.Character:FindFirstChild("HumanoidRootPart") then
            local d = (HRP.Position - p.Character.HumanoidRootPart.Position).Magnitude
            if d < dist then
                dist = d
                best = p
            end
        end
    end
    return best
end

local function DropTarget()
    if CurrentTarget then
        LastDrop[CurrentTarget] = tick()
    end
    CurrentTarget = nil
    if BV then BV.Velocity = Vector3.zero end
end

-- ================= SQUARE DASH OFFSET =================
local phase = 0
local lastStep = 0

local function SquareOffset(dist)
    -- vuông to khi xa, nhỏ khi gần
    local r = math.clamp(dist / 30, 1.2, 4)

    if tick() - lastStep > CFG.SquareStepTime then
        phase = (phase + 1) % 4
        lastStep = tick()
    end

    if phase == 0 then
        return Vector3.new( r, CFG.HeadHeight,  0)
    elseif phase == 1 then
        return Vector3.new( 0, CFG.HeadHeight,  r)
    elseif phase == 2 then
        return Vector3.new(-r, CFG.HeadHeight,  0)
    else
        return Vector3.new( 0, CFG.HeadHeight, -r)
    end
end

-- ================= MAIN LOOP =================
RunService.Heartbeat:Connect(function(dt)
    if not getgenv().AutoFollow or not HRP or not BV then return end

    if CurrentTarget and InvalidTarget(CurrentTarget) then
        DropTarget()
    end

    if not CurrentTarget then
        CurrentTarget = GetTarget()
        return
    end

    local THRP = CurrentTarget.Character and CurrentTarget.Character:FindFirstChild("HumanoidRootPart")
    if not THRP then
        DropTarget()
        return
    end

    local dist = (HRP.Position - THRP.Position).Magnitude
    local speed =
        dist > CFG.MidDist and CFG.FastSpeed
        or dist > CFG.Near and CFG.MidSpeed
        or CFG.SlowSpeed

    local offset = SquareOffset(dist)
    local targetPos = THRP.Position + offset
    local dir = targetPos - HRP.Position

    if dir.Magnitude > 1 then
        BV.Velocity = dir.Unit * speed
    else
        BV.Velocity = Vector3.zero
    end

    BG.CFrame = CFrame.lookAt(HRP.Position, THRP.Position)
end)
