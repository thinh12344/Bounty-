-- Module 1: Player Scanner & Target Selection
local PlayerScanner = {}
function PlayerScanner:GetPlayers()
    local players = {}
    for _, player in pairs(game:GetService("Players"):GetPlayers()) do
        if player ~= game.Players.LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local char = player.Character
            local hrp = char.HumanoidRootPart
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            
            if humanoid and humanoid.Health > 0 then
                local distance = (game.Players.LocalPlayer.Character.HumanoidRootPart.Position - hrp.Position).Magnitude
                
                -- Get bounty (you'll need to adjust this based on actual game structure)
                local bounty = 0
                local leaderstats = player:FindFirstChild("leaderstats")
                if leaderstats then
                    local bountyStat = leaderstats:FindFirstChild("Bounty") or leaderstats:FindFirstChild("$")
                    if bountyStat then
                        bounty = tonumber(bountyStat.Value) or 0
                    end
                end
                
                table.insert(players, {
                    Player = player,
                    Character = char,
                    HRP = hrp,
                    Humanoid = humanoid,
                    Distance = distance,
                    Bounty = bounty,
                    Health = humanoid.Health
                })
            end
        end
    end
    return players
end

function PlayerScanner:SelectTarget(players)
    if #players == 0 then return nil end
    
    if Config.TargetPriority == "HighestBounty" then
        table.sort(players, function(a, b)
            return a.Bounty > b.Bounty
        end)
    elseif Config.TargetPriority == "Nearest" then
        table.sort(players, function(a, b)
            return a.Distance < b.Distance
        end)
    elseif Config.TargetPriority == "LowestHealth" then
        table.sort(players, function(a, b)
            return a.Health < b.Health
        end)
    end
    
    return players[1]
end

-- Module 2: Skill Manager
local SkillManager = {}
function SkillManager:GetEquippedTool()
    local player = game.Players.LocalPlayer
    local backpack = player:FindFirstChild("Backpack")
    local character = player.Character
    
    if character then
        for _, tool in pairs(character:GetChildren()) do
            if tool:IsA("Tool") then
                return tool.Name
            end
        end
    end
    
    if backpack then
        for _, tool in pairs(backpack:GetChildren()) do
            if tool:IsA("Tool") then
                return tool.Name
            end
        end
    end
    
    return nil
end

function SkillManager:UseSkill(key)
    local virtualInput = game:GetService("VirtualInputManager")
    virtualInput:SendKeyEvent(true, key, false, game)
    task.wait(0.05)
    virtualInput:SendKeyEvent(false, key, false, game)
end

function SkillManager:ExecuteCombo(target)
    if not target or not target.HRP then return end
    
    -- Face target
    game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.lookAt(
        game.Players.LocalPlayer.Character.HumanoidRootPart.Position,
        target.HRP.Position
    )
    
    -- Check equipped items and use skills
    local toolName = self:GetEquippedTool()
    if toolName then
        -- Sword skills
        if string.find(toolName:lower(), "sword") and Config.UseSword then
            for _, skill in pairs(SwordSkills) do
                self:UseSkill(skill)
                task.wait(Config.SkillComboDelay)
            end
        end
        
        -- Gun skills
        if string.find(toolName:lower(), "gun") and Config.UseGun then
            self:UseSkill("Z")
            task.wait(0.5)
        end
    end
    
    -- Devil Fruit skills (check if fruit is equipped)
    if Config.UseDevilFruit then
        -- Check if player has devil fruit abilities
        local character = game.Players.LocalPlayer.Character
        local hasFruit = false
        
        -- This detection logic needs adjustment based on actual game
        for _, v in pairs(character:GetChildren()) do
            if v:IsA("Tool") and (v.Name:find("Fruit") or v.Name:find("Devil")) then
                hasFruit = true
                break
            end
        end
        
        if hasFruit then
            for _, skill in pairs(DevilFruitSkills) do
                self:UseSkill(skill)
                task.wait(Config.SkillComboDelay)
            end
        end
    end
    
    -- Fighting style skills
    if Config.UseFightingStyle then
        for _, skill in pairs(FightingSkills) do
            self:UseSkill(skill)
            task.wait(Config.SkillComboDelay)
        end
    end
end

-- Module 3: Flight System
local FlightSystem = {}
function FlightSystem:EnableFlight()
    local player = game.Players.LocalPlayer
    local character = player.Character
    if not character then return end
    
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    -- Create flight body velocity
    local flight
    if not character:FindFirstChild("DarkForgeFlight") then
        flight = Instance.new("BodyVelocity")
        flight.Name = "DarkForgeFlight"
        flight.MaxForce = Vector3.new(100000, 100000, 100000)
        flight.Velocity = Vector3.new(0, 0, 0)
        flight.P = 100000
        flight.Parent = hrp
    end
    
    -- Enable noclip
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
        end
    end
    
    return flight
end

function FlightSystem:FlyTo(position, speed)
    local flight = self:EnableFlight()
    if not flight then return false end
    
    local hrp = game.Players.LocalPlayer.Character.HumanoidRootPart
    local direction = (position - hrp.Position).Unit
    
    flight.Velocity = direction * speed
    
    -- Calculate height adjustment for better flight
    local heightDifference = position.Y - hrp.Position.Y
    if math.abs(heightDifference) > 5 then
        flight.Velocity = flight.Velocity + Vector3.new(0, heightDifference * 0.5, 0)
    end
    
    return true
end

function FlightSystem:StopFlight()
    local character = game.Players.LocalPlayer.Character
    if character then
        local flight = character:FindFirstChild("DarkForgeFlight")
        if flight then
            flight:Destroy()
        end
    end
end

function ESPLibrary:CreateESP(player, data)
    if not Config.ESPEnabled then return end
    
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    if not ESPLibrary.ESPObjects[player] then
        ESPLibrary.ESPObjects[player] = {}
    end
    
    -- Create Billboard GUI for name and bounty
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "DF_ESP"
    billboard.AlwaysOnTop = true
    billboard.Size = UDim2.new(0, 200, 0, 50)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.Adornee = character.HumanoidRootPart
    billboard.Parent = character.HumanoidRootPart
    
    local nameLabel = Instance.new("TextLabel")
    nameLabel.Name = "Name"
    nameLabel.Text = player.Name
    nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    nameLabel.TextStrokeTransparency = 0
    nameLabel.BackgroundTransparency = 1
    nameLabel.Size = UDim2.new(1, 0, 0.5, 0)
    nameLabel.Parent = billboard
    
    local bountyLabel = Instance.new("TextLabel")
    bountyLabel.Name = "Bounty"
    bountyLabel.Text = "Bounty: " .. tostring(data.Bounty)
    bountyLabel.TextColor3 = Color3.fromRGB(255, 215, 0)
    bountyLabel.TextStrokeTransparency = 0
    bountyLabel.BackgroundTransparency = 1
    bountyLabel.Size = UDim2.new(1, 0, 0.5, 0)
    bountyLabel.Position = UDim2.new(0, 0, 0.5, 0)
    bountyLabel.Parent = billboard
    
    -- Create box ESP
    if Config.BoxESP then
        local box = Instance.new("BoxHandleAdornment")
        box.Name = "DF_Box"
        box.Adornee = character.HumanoidRootPart
        box.AlwaysOnTop = true
        box.ZIndex = 5
        box.Size = character.HumanoidRootPart.Size * 1.5
        box.Color3 = Color3.fromRGB(255, 0, 0)
        box.Transparency = 0.5
        box.Parent = character.HumanoidRootPart
        
        table.insert(ESPLibrary.ESPObjects[player], box)
    end
    
    -- Create tracer
    if Config.TracerESP then
        local tracer = Drawing.new("Line")
        tracer.Visible = true
        tracer.Color = Color3.fromRGB(255, 50, 50)
        tracer.Thickness = 2
        tracer.Transparency = 1
        
        table.insert(ESPLibrary.ESPObjects[player], tracer)
    end
    
    table.insert(ESPLibrary.ESPObjects[player], billboard)
end

function ESPLibrary:UpdateESP()
    for player, objects in pairs(ESPLibrary.ESPObjects) do
        if not player or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
            ESPLibrary:RemoveESP(player)
        else
            -- Update tracer position
            for _, obj in pairs(objects) do
                if obj.ClassName == "Line" then
                    local screenPos, visible = game.Workspace.CurrentCamera:WorldToViewportPoint(
                        player.Character.HumanoidRootPart.Position
                    )
                    
                    if visible then
                        obj.Visible = true
                        obj.From = Vector2.new(
                            game.Workspace.CurrentCamera.ViewportSize.X / 2,
                            game.Workspace.CurrentCamera.ViewportSize.Y
                        )
                        obj.To = Vector2.new(screenPos.X, screenPos.Y)
                    else
                        obj.Visible = false
                    end
                end
            end
        end
    end
end

function ESPLibrary:RemoveESP(player)
    if ESPLibrary.ESPObjects[player] then
        for _, obj in pairs(ESPLibrary.ESPObjects[player]) do
            if obj.ClassName == "Line" then
                obj:Remove()
            else
                obj:Destroy()
            end
        end
        ESPLibrary.ESPObjects[player] = nil
    end
end

function ESPLibrary:ClearAllESP()
    for player, _ in pairs(ESPLibrary.ESPObjects) do
        ESPLibrary:RemoveESP(player)
    end
end

-- Main farming function
local function FarmLoop()
    while IsRunning and Config.AutoFarm do
        -- Scan for players
        local players = PlayerScanner:GetPlayers()
        
        -- Update ESP
        for _, playerData in pairs(players) do
            if not ESPLibrary.ESPObjects[playerData.Player] then
                ESPLibrary:CreateESP(playerData.Player, playerData)
            end
        end
        ESPLibrary:UpdateESP()
        
        -- Select target
        CurrentTarget = PlayerScanner:SelectTarget(players)
        
        -- Update GUI
        if StatusLabel then
            StatusLabel.Text = "Status: " .. (#players > 0 and "HUNTING" or "SEARCHING")
        end
        
        if TargetLabel and CurrentTarget then
            TargetLabel.Text = string.format("Target: %s | Bounty: %d | Distance: %d",
                CurrentTarget.Player.Name,
                CurrentTarget.Bounty,
                math.floor(CurrentTarget.Distance)
            )
        end
        
        -- If we have a target
        if CurrentTarget and CurrentTarget.Distance < Config.MaxDistance then
            -- Fly to target
            if Config.AutoFlight and CurrentTarget.Distance > 30 then
                FlightSystem:FlyTo(CurrentTarget.HRP.Position, Config.FlightSpeed)
            else
                FlightSystem:StopFlight()
            end
            
            -- Use skills if in range
            if CurrentTarget.Distance < 50 then
                SkillManager:ExecuteCombo(CurrentTarget)
                
                -- Check if target died
                if not CurrentTarget.Humanoid or CurrentTarget.Humanoid.Health <= 0 then
                    TotalBountyFarmed = TotalBountyFarmed + CurrentTarget.Bounty
                    if BountyLabel then
                        BountyLabel.Text = "Total Bounty Farmed: " .. TotalBountyFarmed
                    end
                    task.wait(1) -- Wait for bounty update
                end
            end
        else
            FlightSystem:StopFlight()
        end
        
        task.wait(0.1) -- Main loop delay
    end
end

local SkillLock = {}
function SkillLock:LockOnTarget(target)
    if not target or not target.HRP then return end
    
    -- Calculate prediction for moving targets
    local hrp = target.HRP
    local velocity = hrp.Velocity
    local distance = (LocalPlayer.Character.HumanoidRootPart.Position - hrp.Position).Magnitude
    
    -- Prediction time based on distance and projectile speed
    local travelTime = distance / 100 -- Adjust based on skill speed
    local predictedPosition = hrp.Position + (velocity * travelTime)
    
    -- Aim at predicted position
    LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.lookAt(
        LocalPlayer.Character.HumanoidRootPart.Position,
        predictedPosition
    )
    
    return predictedPosition
end

end

-- Feature: Health monitor & auto-run
local SafetySystem = {}
function SafetySystem:CheckHealth()
    local character = LocalPlayer.Character
    if not character then return true end
    
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return true end
    
    if humanoid.Health < humanoid.MaxHealth * 0.3 then
        -- Low health, run away
        FlightSystem:FlyTo(
            character.HumanoidRootPart.Position + Vector3.new(math.random(-100, 100), 50, math.random(-100, 100)),
            Config.FlightSpeed * 1.5
        )
        return false
    end
    
    return true
end
